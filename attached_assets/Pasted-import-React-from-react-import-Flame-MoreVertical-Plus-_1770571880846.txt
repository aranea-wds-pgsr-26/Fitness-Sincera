/**
 * Componente que renderiza os blocos individuais do Gauge usando matemática polar
 */
const GaugeBlocks = ({ totalBlocks, progressBlocks, color }) => {
  const blocks = [];
  const startAngle = 180;
  const endAngle = 360;
  const radius = 90;
  const centerX = 120;
  const centerY = 130;
  
  // Se progressBlocks for 0, desenhamos o fundo total
  const limit = progressBlocks > 0 ? progressBlocks : totalBlocks;

  for (let i = 0; i < limit; i++) {
    const angle = startAngle + (i * (endAngle - startAngle)) / (totalBlocks - 1);
    const rad = (angle * Math.PI) / 180;
    
    // Coordenadas para blocos rectangulares rotacionados
    const x = centerX + radius * Math.cos(rad);
    const y = centerY + radius * Math.sin(rad);
    
    blocks.push(
      <rect
        key={i}
        x={x - 4}
        y={y - 10}
        width="8"
        height="20"
        rx="4"
        fill={color}
        transform={`rotate(${angle + 90}, ${x}, ${y})`}
      />
    );
  }
  
  return <g>{blocks}</g>;
};

};
